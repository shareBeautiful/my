import pymysql

# host：数据库地址，user：用户名，passwor：密码，port：端口，db：数据库名
conn = pymysql.connect(host='localhost', user='root', password='123456', port=3306, db='mysql')

# cursor() 数据库操作对象
cursor = conn.cursor()
# execute 执行方法
cursor.execute('select * from db')
# 
cursor.fetchone()

# 查询
